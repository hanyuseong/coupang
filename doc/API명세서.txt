0. Auth / Login / Signup
🔹 POST /api/auth/signup – 회원가입
Request
{
  "email": "user@example.com",
  "password": "1234!abcd",
  "name": "홍길동",
  "phone": "01012345678"
}

Response
{
  "memberId": 1,
  "email": "user@example.com"
}

🔹 POST /api/auth/login
Request
{
  "email": "user@example.com",
  "password": "1234!abcd"
}

Response
{
  "accessToken": "JWT-ACCESS-TOKEN",
  "refreshToken": "JWT-REFRESH-TOKEN"
}

🔹 POST /api/auth/refresh
Request
{
  "refreshToken": "xxxxx"
}

Response
{
  "accessToken": "new-access-token"
}

}
